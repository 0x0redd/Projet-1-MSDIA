\section{Contexte}

\begin{frame}{1) Périmètre du projet}
\begin{itemize}
  \item Objectif: classifier des IRM cérébrales avec des méthodes classiques de Machine Learning.
  \item Approche: ingénierie de caractéristiques + benchmark systématique des modèles.
  \item Modèles évalués: \textbf{SVM}, \textbf{k-NN}, \textbf{Random Forest}.
  \item Message clé: une pipeline ML classique bien conçue reste compétitive et explicable.

\begin{frame}{2) Problématique et motivation}
\begin{itemize}
  \item Données image: forte dimension, variabilité intra-classe, bruit, illumination.
  \item Défi principal: transformer les pixels bruts en représentations discriminantes.
  \item Contraintes: reproductibilité, robustesse, coût de calcul maîtrisé.
  \item Critères de succès: Accuracy/F1, stabilité (CV + seeds), compromis performance/temps.
\end{itemize}
\end{frame}

\begin{frame}{5) Pourquoi des métriques de séparabilité en Phase 1 ?}
\begin{itemize}
  \item Entraîner un classifieur sur chaque combinaison serait trop coûteux.
  \item On évalue rapidement la qualité intrinsèque des descripteurs via:
  \begin{itemize}
    \item \textbf{FDR} (plus grand = meilleur),
    \item \textbf{MI} (plus grand = meilleur),
    \item \textbf{DBI} (plus petit = meilleur).
  \end{itemize}
  \item Score composite utilisé pour classer les configurations:
  \[
  \text{score}=\frac{FDR_n + MI_n + (1-DBI_n)}{3}
  \]
\end{itemize}
\end{frame}

\section{Benchmark Phase 2}

\begin{frame}{10) Design du benchmark}
\begin{itemize}
  \item 8 jeux de caractéristiques: blocs unitaires, fusions partielles, fusion complète \texttt{Full(opt)}.
  \item Trois modèles comparés: SVM, k-NN, Random Forest.
  \item Recherche d'hyperparamètres pour chaque couple (feature set, modèle).
  \item Évaluation en mode \texttt{split} + multi-CV (\texttt{cv\_1} à \texttt{cv\_5}).
\end{itemize}
\vspace{0.3em}
\footnotesize Sources: \texttt{phase2\_benchmark\_results.csv}, \texttt{phase2\_all\_modes\_metrics.csv}.
\end{frame}

\begin{frame}{11) Focus -- Best feature parameters et évaluation}
\small
\begin{itemize}
  \item Les jeux \texttt{(opt)} utilisent les meilleurs paramètres de Phase 1:
  \texttt{B -- Tex(opt), C -- DWT(opt), B+C(opt), A+B+C(opt), Full(opt)}.
  \item Meilleur résultat global: \textbf{Full(opt)+SVM} (Acc=0.9067, F1-macro=0.9064).
  \item Très proche: \textbf{D -- HOG+SVM} (Acc=0.9033, F1-macro=0.9030).
  \item \textbf{Full(opt)+RF}: Acc=0.8700, F1=0.8697.
  \item \textbf{Full(opt)+KNN}: Acc=0.8433, F1=0.8405.
\end{itemize}
\textbf{Message:} l'optimisation des descripteurs + la fusion multi-caractéristiques maximisent la performance.
\end{frame}

\begin{frame}{12) Classement global des configurations}
\begin{center}
  \includegraphics[width=0.82\linewidth]{output/p2_leaderboard.png}
\end{center}
\begin{itemize}
  \item Gagnant clair: \texttt{Full(opt)+SVM}.
  \item SVM domine les premières places, RF offre un compromis robuste.
\end{itemize}
\end{frame}

\begin{frame}{13) Focus -- Hyperparamètres et combinaisons entraînées}
\small
\begin{itemize}
  \item \textbf{SVM}: majoritairement RBF, \(C \in \{1,10,100\}\), \(\gamma \in \{\text{scale}, 0.01\}\).
  \item \textbf{k-NN}: \(n\_neighbors \in \{3,9,11\}\), distances Manhattan/Euclidean, pondération distance fréquente.
  \item \textbf{RF}: \(n\_estimators\) 200--300, \(max\_features \in \{sqrt,log2\}\), profondeur 10 ou None.
  \item Les variations \texttt{split} + \texttt{cv\_1..cv\_5} confirment les tendances, sans optimum fragile unique.
\end{itemize}
\end{frame}

\begin{frame}{14) Trade-off performance / coût d'entraînement}
\begin{itemize}
  \item \textbf{Meilleure performance absolue}: \texttt{Full(opt)+SVM}
  \begin{itemize}
    \item Acc=0.9067, temps d'entraînement \(\sim 696s\)
  \end{itemize}
  \item \textbf{Très haute performance}: \texttt{D -- HOG+SVM}
  \begin{itemize}
    \item Acc=0.9033, temps \(\sim 709s\)
  \end{itemize}
  \item \textbf{Option plus rapide}: \texttt{A+B+RF}
  \begin{itemize}
    \item Acc=0.8300, temps \(\sim 8.29s\)
  \end{itemize}
\end{itemize}
\textbf{Conclusion:} cartographie complète du compromis précision--temps.
\end{frame}

\begin{frame}{5) Why Phase 1 Uses Separability Metrics}
\begin{itemize}
  \item Exhaustive model training for every parameter set is expensive
  \item Phase 1 uses fast classifier-free metrics:
  \begin{itemize}
    \item FDR (higher is better)
    \item MI (higher is better)
    \item DBI (lower is better)
  \end{itemize}
  \item Composite score:
  \[
    \text{score}=\frac{\mathrm{FDR}_n + \mathrm{MI}_n + (1-\mathrm{DBI}_n)}{3}
  \]
\end{itemize}
